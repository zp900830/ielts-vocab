# 「走完全部词要多久」实现计划

> **For agentic workers:** REQUIRED SUB-SKILL: 用 superpowers:subagent-driven-development 或 superpowers:executing-plans 逐任务实现。步骤用 `- [ ]` 勾选跟踪。

**Goal:** 让界面在**初次设置、计划页、今天面板注脚**三处用同一套算法告诉他「照每天 N 分钟，全部毕业约在某年某月（还需 M 天）」，并给 ②/③ 的模拟提供一条不漂移的算式。

**Architecture:** 引擎侧新增一个纯函数 `estimateDays(state, minutes, opts)` —— 逐日调**现成的** `assemble` 排队列 → 把队列翻成**现成的** `mkContact`/`mkQuiz` 事件 → 只把新事件续算进同一份 state（`replay` 新增 `opts.state`）。界面三处只调它一次并共用一个格式化器，**任何地方不许写"词数×20÷每天句数"这类旁路公式**。

**Tech Stack:** 无构建、无依赖的静态站。引擎 `shadow/js/plan-engine.js` 是裸 IIFE 挂 `window.ShadowPlan`（**Node 里 require 不了**）；测试全部走 Playwright，在浏览器里用 `page.evaluate` 直接读全局。

**Spec:** `docs/superpowers/specs/2026-09-21-走完全部词要多久-设计.md`（**规格与本计划冲突时以规格为准，但要回来告诉我**，别自己改口径）

## Global Constraints

每条都适用于所有任务，值逐字抄自规格/仓库：

- [ ] **Step 3: 实现**

插到 `shadow/js/plan-engine.js` 里 `assemble` 函数结束（`}` 在 273 行附近）与 `recallQuiz`（283 行附近）之间：

```js
  /* 已毕业词数。引擎里没有现成计数器（daily 只记 promote/minutes 这类过程量），
     所以只能遍历 —— 3245 个槽一天一次，量级 3ms，别为它加字段（加了就多一份真值）。 */
  function countGraduated(st) {
    let n = 0;
    const ws = (st && st.words) || {};
    for (const k in ws) if (ws[k].stage === 'graduated') n++;
    return n;
  }

  /* 「照每天 N 分钟，全部毕业大约要多久」—— 全项目唯一一份算式，界面三处都调这里。
     做法是照真流程走一遍：逐日用现成 assemble 排队列 → 队列翻成 contact/quiz 事件
     → 只把新事件续算进同一份 state（Task 1 的 opts.state）→ 数毕业词。
     两条不能省：
       ① 必须深拷 state。replay 会原地改它，直接喂活状态等于把他真实进度算坏。
       ② 不许在这里另写"每天几个词"的公式。宁可慢，也不能有两份真值。 */
  function estimateDays(state, minutes, opts) {
    const o = opts || {};
    const boundary = Number.isInteger(o.boundaryHour) ? o.boundaryHour : 4;
    const total = o.totalSents || 0;
    const wordsOf = typeof o.wordsOf === 'function' ? o.wordsOf : function () { return []; };
    const accuracy = typeof o.accuracy === 'number' ? o.accuracy : 0.85;
    const maxDays = o.maxDays || 1095;
    const horizon = o.horizonDays || 365;
    const start = Number.isFinite(o.now) ? o.now : Date.now();
    const allWords = o.totalWords || 0;
    const plan = o.plan || (state && state.plan) || null;
    const asOpts = { totalSents: total, wordsOf: wordsOf, boundaryHour: boundary, plan: plan,
                     secNew: o.secNew, secReview: o.secReview, secQuiz: o.secQuiz, rate: o.rate };
    let st = JSON.parse(JSON.stringify(state || emptyState()));
    st.plan = plan;
    const graduatedNow = countGraduated(st);
    let daysUsed = 0, done = allWords > 0 && graduatedNow >= allWords, atHorizon = null, empty = false;
    for (let day = 0; !done && day <= maxDays; day++) {
      const now = start + day * DAY_MS;
      const dayName = dayKey(now, boundary);
      if (day === horizon) atHorizon = { days: horizon, graduated: countGraduated(st), at: now };
      asOpts.now = now; asOpts.todayMinutes = minutes;
      const r = assemble(st, asOpts);
      const fresh = [];
      for (let i = 0; i < r.queue.length; i++) {
        const q = r.queue[i];
        for (let j = 0; j < q.words.length; j++) fresh.push(mkContact(q.words[j], q.i, now, dayName));
      }
      for (let i = 0; i < r.items.length; i++) {
        const it = r.items[i];
        const kind = it.pass === 3 ? 'mc4zh' : 'recall';
        // 确定性抽签：同一个 (天, 词) 永远同结果。换成 Math.random 会让上面的单调性测试随机红。
        const ok = (hash32(day * 7919 + it.w.length * 131 + (it.s === 'ex' ? 7 : it.s)) % 1000) < accuracy * 1000;
        fresh.push(mkQuiz(it.w, it.s, kind, ok, now, dayName));
      }
      if (!fresh.length) { empty = true; break; }   // 今天一个都排不出来 → 再等下去也不会毕业，别再空转
      st = replay(fresh, { state: st, boundaryHour: boundary, wordsOf: wordsOf, plan: plan });
      if (allWords > 0 && countGraduated(st) >= allWords) { done = true; daysUsed = day + 1; }
    }
    return {
      graduatedNow: graduatedNow, total: allWords,
      done: done, days: done ? daysUsed : null, doneAt: done ? start + daysUsed * DAY_MS : null,
      atHorizon: atHorizon, capped: !done && !empty, empty: empty,
    };
  }
```

        const st = ROOT2.state || {};

        if (_etaMemo && _etaMemo.key === key) return Promise.resolve(_etaMemo.val);

              boundaryHour: (ROOT2.plan && ROOT2.plan.boundaryHour) || 4, plan: ROOT2.plan,
              secNew: secNew(), secReview: secReview(), secQuiz: secQuiz(),
            });

      function etaMonth(ts, exact) {
        const d = new Date(ts);
        return d.getFullYear() + ' 年 ' + (d.getMonth() + 1) + ' 月' + (exact ? ' ' + d.getDate() + ' 日' : '');
      }
      function etaText(r, minutes, exact) {                       // 规格 §4.2 那三种封顶/失败话术
        if (!r || r.empty) return '按每天 ' + minutes + ' 分钟排不出任务，先调大一点或取消「只复习」';
        if (!r.done) return '按每天 ' + minutes + ' 分钟，三年内到不了终点（先把每天时间调大些）';
        return '到 ' + etaMonth(r.doneAt, exact) + ' 全部毕业（需要 ' + r.days + ' 天）';
      }
      function milestoneText(r) {
        if (!r || !r.atHorizon) return '';
        return '到 ' + etaMonth(r.atHorizon.at, false) + ' 能毕业约 ' + r.atHorizon.graduated + ' 个词';
      }
      function activeDaysRecent(n) {
        const d = (ROOT2.state && ROOT2.state.daily) || {};
        const bd = (ROOT2.plan && ROOT2.plan.boundaryHour) || 4;
        let c = 0;
        for (let i = 0; i < n; i++) {
          const x = d[ShadowPlan.dayKey(Date.now() - i * 864e5, bd)];
          if (x && ((x.sentDone || 0) > 0 || (x.minutes || 0) > 0)) c++;
        }
        return c;
      }
