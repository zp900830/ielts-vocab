```js
      const BACKUP_PREFIX = 'ielts-';
      /* 点分隔的 key 不受 BACKUP_PREFIX（连字符）覆盖，逐个显式列出。LS_ARTICLE 是 3.0 的
         按篇账（§7.6/§9.4），必须随备份走 —— 否则换设备后精读次数与「已学完」凭据全丢。 */
      const BACKUP_EXTRA = ['ielts.shadow.v2', LS_ARTICLE];
      const isBackupKey = k => !!k && (k.startsWith(BACKUP_PREFIX) || BACKUP_EXTRA.includes(k));
      /* 这两个根必须是能 JSON.parse 的对象：坏值会让下次启动的引擎重放直接抛错（白屏）。
         其余 ielts-* 是标量（音色 URI / 深色标记 / 倍速…），只要求是字符串。 */
      const BACKUP_JSON_ROOTS = ['ielts.shadow.v2', LS_ARTICLE];
      function collectBackup() {
        const data = {};
        for (let i = 0; i < localStorage.length; i++) {
          const k = localStorage.key(i);
          if (isBackupKey(k)) data[k] = localStorage.getItem(k);
        }
        return data;
      }
      function validBackupValue(k, v) {
        if (typeof v !== 'string') return false;
        if (BACKUP_JSON_ROOTS.includes(k)) {
          try { const o = JSON.parse(v); return !!o && typeof o === 'object'; } catch (e) { return false; }
        }
        return true;
      }
      /* 只校验、不落盘 —— 导入流程在 confirm 之前先跑它，坏备份根本不进写入阶段。 */
      function parseBackup(o) {
        if (!o || typeof o !== 'object' || Array.isArray(o)) return { ok: false, reason: '备份文件格式不对' };
        if (o.app !== 'ielts-shadow' || !o.data || typeof o.data !== 'object' || Array.isArray(o.data)) {
          return { ok: false, reason: '这不是影子跟读导出的备份文件' };
        }
        const keys = Object.keys(o.data).filter(isBackupKey);
        if (!keys.length) return { ok: false, reason: '备份里没有可恢复的数据' };
        for (const k of keys) {
          if (!validBackupValue(k, o.data[k])) return { ok: false, reason: `备份里的「${k}」内容已损坏，已拒绝导入` };
        }
        return { ok: true, keys };
      }
      function backupPayload() {
        return { app: 'ielts-shadow', backupVer: 1, exportedAt: new Date().toISOString(), data: collectBackup() };
      }
      /* 落盘：先快照整个备份域，任一步失败就把域内清空 + 按快照写回 —— 绝不半截覆盖。
         返回 {ok} 或 {ok:false,reason,rolledBack}；不 reload、不 confirm（UX 由 importBackup 负责）。 */
      function applyBackup(o, keys) {
        const snapshot = collectBackup();
        const wanted = keys || Object.keys(o.data).filter(isBackupKey);
        try {
          wanted.forEach(k => localStorage.setItem(k, o.data[k]));
          Object.keys(collectBackup()).forEach(k => { if (!wanted.includes(k)) localStorage.removeItem(k); });
        } catch (e) {
          try {
            Object.keys(collectBackup()).forEach(k => localStorage.removeItem(k));
            Object.keys(snapshot).forEach(k => localStorage.setItem(k, snapshot[k]));
          } catch (e2) {}
          return { ok: false, reason: '导入失败：本地存储写入被拒绝（可能已满），已回滚', rolledBack: true };
        }
        return { ok: true };
      }
      function exportBackup() {
        const payload = backupPayload();
        const blob = new Blob([JSON.stringify(payload, null, 2)], { type: 'application/json' });
        const a = document.createElement('a');
        a.href = URL.createObjectURL(blob);
        const d = new Date();
        const stamp = d.getFullYear() + String(d.getMonth() + 1).padStart(2, '0') + String(d.getDate()).padStart(2, '0');
        a.download = `影子跟读备份_${stamp}.json`;
        document.body.appendChild(a);
        a.click();
        a.remove();
        setTimeout(() => URL.revokeObjectURL(a.href), 4000);
        showAbToast(`已导出 ${Object.keys(payload.data).length} 项本地数据，请妥善保存文件`);
      }
      function importBackup(id) {
        const f = document.getElementById(id || 'importFile');
        if (!f) return;
        f.onchange = async () => {
          const file = f.files && f.files[0];
          f.value = '';
          if (!file) return;
          let o;
          try { o = JSON.parse(await file.text()); } catch (e) { showAbToast('导入失败：文件不是有效 JSON'); return; }
          const parsed = parseBackup(o);
          if (!parsed.ok) { showAbToast('导入失败：' + parsed.reason); return; }
          if (!confirm(`导入将覆盖当前全部跟读数据（计划、进度、书签、播放位）。\n备份导出时间：${o.exportedAt || '未知'}\n确定继续吗？`)) return;
          const r = applyBackup(o, parsed.keys);
          if (!r.ok) { showAbToast(r.reason); return; }
          showAbToast('导入成功，正在重新加载');
          setTimeout(() => location.reload(), 400);
        };
        f.click();
      }
```

```js
    pop.addEventListener('keydown', (e) => {
      if (e.key !== 'Tab') return;
      const nodes = Array.prototype.filter.call(
        pop.querySelectorAll('button, [href], input, select, textarea, [tabindex]:not([tabindex="-1"])'),
        (el) => !el.disabled && el.offsetParent !== null);
      if (!nodes.length) return;
      const first = nodes[0], last = nodes[nodes.length - 1];
      if (e.shiftKey && (document.activeElement === first || document.activeElement === pop)) {
        e.preventDefault(); last.focus();
      } else if (!e.shiftKey && document.activeElement === last) {
        e.preventDefault(); first.focus();
      }
    });
```
